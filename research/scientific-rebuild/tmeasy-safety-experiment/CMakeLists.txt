cmake_minimum_required(VERSION 3.20)
project(tmeasy_safety_experiment LANGUAGES CXX)
if(NOT CHRONO_SOURCE OR NOT CHRONO_BUILD OR NOT VEHICLE_LIBRARY_DIR OR NOT EIGEN_INCLUDE)
  message(FATAL_ERROR "Supply original pinned source/build, selected Vehicle library directory, and Eigen include")
endif()
foreach(probe IN ITEMS frozen trajectory)
  if(probe STREQUAL frozen)
    set(probe_source ../tmeasy-native-verification/native.cpp)
  else()
    set(probe_source trajectory.cpp)
  endif()
  add_executable(${probe} ${probe_source}
    ${CHRONO_SOURCE}/src/chrono_models/vehicle/generic/Generic_Wheel.cpp
    ${CHRONO_SOURCE}/src/chrono_models/vehicle/generic/tire/Generic_TMeasyTire.cpp)
  target_compile_features(${probe} PRIVATE cxx_std_17)
  target_compile_options(${probe} PRIVATE -march=native)
  target_compile_definitions(${probe} PRIVATE EIGEN_DONT_PARALLELIZE _ENABLE_EXTENDED_ALIGNED_STORAGE)
  target_include_directories(${probe} PRIVATE ${CHRONO_SOURCE}/src ${CHRONO_BUILD} ${EIGEN_INCLUDE})
  target_link_libraries(${probe} PRIVATE ${VEHICLE_LIBRARY_DIR}/libChrono_vehicle.so ${CHRONO_BUILD}/lib/libChrono_core.so)
  set_target_properties(${probe} PROPERTIES BUILD_RPATH "${VEHICLE_LIBRARY_DIR};${CHRONO_BUILD}/lib")
endforeach()
