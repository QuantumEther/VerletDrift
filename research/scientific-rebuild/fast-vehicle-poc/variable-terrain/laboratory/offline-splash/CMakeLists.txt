cmake_minimum_required(VERSION 3.10)
project(VerletOfflineSplash)
if(NOT SPLISH_SOURCE)
 message(FATAL_ERROR "Set SPLISH_SOURCE to the pinned SPlisHSPlasH checkout")
endif()
set(SPH_LIBS_ONLY ON CACHE BOOL "" FORCE)
set(USE_THIRD_PARTY_METHODS OFF CACHE BOOL "" FORCE)
add_subdirectory(${SPLISH_SOURCE} splish)
add_executable(verlet_export smoke.cpp)
target_include_directories(verlet_export PRIVATE ${SPLISH_SOURCE})
target_link_libraries(verlet_export SPlisHSPlasH Utilities)
add_executable(verlet_impact impact.cpp)
target_include_directories(verlet_impact PRIVATE ${SPLISH_SOURCE})
target_link_libraries(verlet_impact SPlisHSPlasH Utilities)
